function doGet(e) { var action = e.parameter.action; // Trỏ chính xác vào sheet name: checkin var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("checkin"); var data = sheet.getDataRange().getValues(); // 1. TẢI DỮ LIỆU BÁO CÁO CHO TAB 2 (DASHBOARD) if (action === "getDashboard") { var checkedIn = []; var notCheckedIn = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (!row[0]) continue; // Bỏ qua dòng trống tên var item = { hoTen: row[0], // Cột A msnv: row[1], // Cột B sdt: row[2], // Cột C zone: row[3], // Cột D soAo: row[4], // Cột E tenDoi: row[5], // Cột F viTri: row[6], // Cột G status: row[7] // Cột H }; if (item.status === "Đã check-in") { checkedIn.push(item); } else { notCheckedIn.push(item); } } return responseJSON({ checkedIn: checkedIn, notCheckedIn: notCheckedIn }); } // 2. XỬ LÝ TRA CỨU & CHECK-IN var query = e.parameter.query; if (!query) return responseJSON({ success: false, message: "Thiếu thông tin tra cứu!" }); query = query.toString().trim().toLowerCase(); for (var i = 1; i < data.length; i++) { var row = data[i]; if (!row[0]) continue; var msnv = row[1] ? row[1].toString().trim().toLowerCase() : ""; var sdt = row[2] ? row[2].toString().trim().toLowerCase() : ""; // Tìm kiếm khớp với MSNV hoặc SĐT if ((msnv && query === msnv) || (sdt && query === sdt)) { // Ghi chữ "Đã check-in" vào Cột H (Cột số 8) sheet.getRange(i + 1, 8).setValue("Đã check-in"); var isBTC = (row[6] === "BTC" || row[3] === "BTC"); return responseJSON({ success: true, data: { hoTen: row[0], msnv: row[1], sdt: row[2], zone: row[3], soAo: row[4], tenDoi: row[5], viTri: row[6], isBTC: isBTC } }); } } return responseJSON({ success: false, message: "Không tìm thấy MSNV hoặc SĐT trong danh sách!" }); } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
